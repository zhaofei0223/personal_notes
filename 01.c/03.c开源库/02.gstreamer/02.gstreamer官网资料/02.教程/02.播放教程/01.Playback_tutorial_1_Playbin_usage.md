# 播放教程 1：Playbin 的使用

## 目标

我们之前已经使用过这个playbin元素，它能够构建完整的播放流程，无需我们进行太​​多额外工作。本教程将展示如何playbin在默认值不符合我们特定需求的情况下进行进一步的自定义。

我们将学习：

- 如何找出文件包含多少个流，以及如何在它们之间切换。
- 如何收集有关各个流的信息。

## 介绍

通常情况下，单个文件中会嵌入多个音频、视频和字幕流。最常见的情况是普通电影，它们通常包含一个视频流和一个音频流（立体声或 5.1 声道音轨被视为一个流）。此外，为了适应不同的语言，包含一个视频流和多个音频流的电影也越来越常见。在这种情况下，用户选择一个音频流后，应用程序只会播放该音频流，而忽略其他音频流。

为了能够选择合适的音频流，用户需要了解一些个人信息，例如他们的语言。这些信息以“元数据”（附加数据）的形式嵌入在音频流中，本教程将演示如何检索这些信息。

字幕可以像音频和视频一样嵌入到文件中，但更详细的讲解请参见“播放教程 2：字幕管理”。最后，单个文件中也可能包含多个视频流，例如，DVD 中同一场景的多个角度，但这种情况比较少见。

    将多个流嵌入到单个文件中称为“多路复用”或“混流”，这样的文件就被称为“容器”。常见的容器格式有 Matroska (.mkv)、Quicktime (.qt、.mov、.mp4)、Ogg (.ogg) 或 Webm (.webm)。
    从容器内检索各个流的过程称为“解复用”或“解复用”。

以下代码恢复文件中的流数量、它们关联的元数据，并允许在媒体播放时切换音频流。

```c
#include <stdio.h>
#include <gst/gst.h>

#ifdef __APPLE__
#include <TargetConditionals.h>
#endif

/* Structure to contain all our information, so we can pass it around */
typedef struct _CustomData
{
  GstElement *playbin;          /* Our one and only element */

  gint n_video;                 /* Number of embedded video streams */
  gint n_audio;                 /* Number of embedded audio streams */
  gint n_text;                  /* Number of embedded subtitle streams */

  gint current_video;           /* Currently playing video stream */
  gint current_audio;           /* Currently playing audio stream */
  gint current_text;            /* Currently playing subtitle stream */

  GMainLoop *main_loop;         /* GLib's Main Loop */
} CustomData;

/* playbin flags */
typedef enum
{
  GST_PLAY_FLAG_VIDEO = (1 << 0),       /* We want video output */
  GST_PLAY_FLAG_AUDIO = (1 << 1),       /* We want audio output */
  GST_PLAY_FLAG_TEXT = (1 << 2) /* We want subtitle output */
} GstPlayFlags;

/* Forward definition for the message and keyboard processing functions */
static gboolean handle_message (GstBus * bus, GstMessage * msg,
    CustomData * data);
static gboolean handle_keyboard (GIOChannel * source, GIOCondition cond,
    CustomData * data);

int
tutorial_main (int argc, char *argv[])
{
  CustomData data;
  GstBus *bus;
  GstStateChangeReturn ret;
  gint flags;
  GIOChannel *io_stdin;

  /* Initialize GStreamer */
  gst_init (&argc, &argv);

  /* Create the elements */
  data.playbin = gst_element_factory_make ("playbin", "playbin");

  if (!data.playbin) {
    g_printerr ("Not all elements could be created.\n");
    return -1;
  }

  /* Set the URI to play */
  g_object_set (data.playbin, "uri",
      "https://gstreamer.freedesktop.org/data/media/sintel_cropped_multilingual.webm",
      NULL);

  /* Set flags to show Audio and Video but ignore Subtitles */
  g_object_get (data.playbin, "flags", &flags, NULL);
  flags |= GST_PLAY_FLAG_VIDEO | GST_PLAY_FLAG_AUDIO;
  flags &= ~GST_PLAY_FLAG_TEXT;
  g_object_set (data.playbin, "flags", flags, NULL);

  /* Set connection speed. This will affect some internal decisions of playbin */
  g_object_set (data.playbin, "connection-speed", 56, NULL);

  /* Add a bus watch, so we get notified when a message arrives */
  bus = gst_element_get_bus (data.playbin);
  gst_bus_add_watch (bus, (GstBusFunc) handle_message, &data);

  /* Add a keyboard watch so we get notified of keystrokes */
#ifdef G_OS_WIN32
  io_stdin = g_io_channel_win32_new_fd (fileno (stdin));
#else
  io_stdin = g_io_channel_unix_new (fileno (stdin));
#endif
  g_io_add_watch (io_stdin, G_IO_IN, (GIOFunc) handle_keyboard, &data);

  /* Start playing */
  ret = gst_element_set_state (data.playbin, GST_STATE_PLAYING);
  if (ret == GST_STATE_CHANGE_FAILURE) {
    g_printerr ("Unable to set the pipeline to the playing state.\n");
    gst_object_unref (data.playbin);
    return -1;
  }

  /* Create a GLib Main Loop and set it to run */
  data.main_loop = g_main_loop_new (NULL, FALSE);
  g_main_loop_run (data.main_loop);

  /* Free resources */
  g_main_loop_unref (data.main_loop);
  g_io_channel_unref (io_stdin);
  gst_object_unref (bus);
  gst_element_set_state (data.playbin, GST_STATE_NULL);
  gst_object_unref (data.playbin);
  return 0;
}

/* Extract some metadata from the streams and print it on the screen */
static void
analyze_streams (CustomData * data)
{
  gint i;
  GstTagList *tags;
  gchar *str;
  guint rate;

  /* Read some properties */
  g_object_get (data->playbin, "n-video", &data->n_video, NULL);
  g_object_get (data->playbin, "n-audio", &data->n_audio, NULL);
  g_object_get (data->playbin, "n-text", &data->n_text, NULL);

  g_print ("%d video stream(s), %d audio stream(s), %d text stream(s)\n",
      data->n_video, data->n_audio, data->n_text);

  g_print ("\n");
  for (i = 0; i < data->n_video; i++) {
    tags = NULL;
    /* Retrieve the stream's video tags */
    g_signal_emit_by_name (data->playbin, "get-video-tags", i, &tags);
    if (tags) {
      g_print ("video stream %d:\n", i);
      gst_tag_list_get_string (tags, GST_TAG_VIDEO_CODEC, &str);
      g_print ("  codec: %s\n", str ? str : "unknown");
      g_free (str);
      gst_tag_list_unref (tags);
    }
  }

  g_print ("\n");
  for (i = 0; i < data->n_audio; i++) {
    tags = NULL;
    /* Retrieve the stream's audio tags */
    g_signal_emit_by_name (data->playbin, "get-audio-tags", i, &tags);
    if (tags) {
      g_print ("audio stream %d:\n", i);
      if (gst_tag_list_get_string (tags, GST_TAG_AUDIO_CODEC, &str)) {
        g_print ("  codec: %s\n", str);
        g_free (str);
      }
      if (gst_tag_list_get_string (tags, GST_TAG_LANGUAGE_CODE, &str)) {
        g_print ("  language: %s\n", str);
        g_free (str);
      }
      if (gst_tag_list_get_uint (tags, GST_TAG_BITRATE, &rate)) {
        g_print ("  bitrate: %d\n", rate);
      }
      gst_tag_list_unref (tags);
    }
  }

  g_print ("\n");
  for (i = 0; i < data->n_text; i++) {
    tags = NULL;
    /* Retrieve the stream's subtitle tags */
    g_signal_emit_by_name (data->playbin, "get-text-tags", i, &tags);
    if (tags) {
      g_print ("subtitle stream %d:\n", i);
      if (gst_tag_list_get_string (tags, GST_TAG_LANGUAGE_CODE, &str)) {
        g_print ("  language: %s\n", str);
        g_free (str);
      }
      gst_tag_list_unref (tags);
    }
  }

  g_object_get (data->playbin, "current-video", &data->current_video, NULL);
  g_object_get (data->playbin, "current-audio", &data->current_audio, NULL);
  g_object_get (data->playbin, "current-text", &data->current_text, NULL);

  g_print ("\n");
  g_print
      ("Currently playing video stream %d, audio stream %d and text stream %d\n",
      data->current_video, data->current_audio, data->current_text);
  g_print
      ("Type any number and hit ENTER to select a different audio stream\n");
}

/* Process messages from GStreamer */
static gboolean
handle_message (GstBus * bus, GstMessage * msg, CustomData * data)
{
  GError *err;
  gchar *debug_info;

  switch (GST_MESSAGE_TYPE (msg)) {
    case GST_MESSAGE_ERROR:
      gst_message_parse_error (msg, &err, &debug_info);
      g_printerr ("Error received from element %s: %s\n",
          GST_OBJECT_NAME (msg->src), err->message);
      g_printerr ("Debugging information: %s\n",
          debug_info ? debug_info : "none");
      g_clear_error (&err);
      g_free (debug_info);
      g_main_loop_quit (data->main_loop);
      break;
    case GST_MESSAGE_EOS:
      g_print ("End-Of-Stream reached.\n");
      g_main_loop_quit (data->main_loop);
      break;
    case GST_MESSAGE_STATE_CHANGED:{
      GstState old_state, new_state, pending_state;
      gst_message_parse_state_changed (msg, &old_state, &new_state,
          &pending_state);
      if (GST_MESSAGE_SRC (msg) == GST_OBJECT (data->playbin)) {
        if (new_state == GST_STATE_PLAYING) {
          /* Once we are in the playing state, analyze the streams */
          analyze_streams (data);
        }
      }
    }
      break;
    default:
      break;
  }

  /* We want to keep receiving messages */
  return TRUE;
}

/* Process keyboard input */
static gboolean
handle_keyboard (GIOChannel * source, GIOCondition cond, CustomData * data)
{
  gchar *str = NULL;

  if (g_io_channel_read_line (source, &str, NULL, NULL,
          NULL) == G_IO_STATUS_NORMAL) {
    int index = g_ascii_strtoull (str, NULL, 0);
    if (index < 0 || index >= data->n_audio) {
      g_printerr ("Index out of bounds\n");
    } else {
      /* If the input was a valid audio stream index, set the current audio stream */
      g_print ("Setting current audio stream to %d\n", index);
      g_object_set (data->playbin, "current-audio", index, NULL);
    }
  }
  g_free (str);
  return TRUE;
}

int
main (int argc, char *argv[])
{
#if defined(__APPLE__) && TARGET_OS_MAC && !TARGET_OS_IPHONE
  return gst_macos_main ((GstMainFunc) tutorial_main, argc, argv, NULL);
#else
  return tutorial_main (argc, argv);
#endif
}
```

在本例中，为了演示目的，我们启用了音频和视频，禁用了字幕，其余标志保持默认值（这就是为什么我们在用 g_object_get()覆盖它之前读取标志的当前值的原因g_object_set()）。

```c
/* Set connection speed. This will affect some internal decisions of playbin */
g_object_set (data.playbin, "connection-speed", 56, NULL);
```

在这个例子中，这个属性实际上用处不大。 connection-speed它指示playbin的是网络连接的最大速度，因此，如果服务器上有多个版本的请求媒体，它会选择最合适的版本。这通常与流媒体协议（例如或 ） playbin结合使用。hlsrtsp

```c
/* Create a GLib Main Loop and set it to run */
data.main_loop = g_main_loop_new (NULL, FALSE);
```

为了实现交互，我们将不再手动轮询 GStreamer 总线。取而代之的是，我们创建一个GMainLoopGLib 主循环并使其运行g_main_loop_run()。该函数会阻塞，直到发出返回值才会返回g_main_loop_quit()。在此期间，它会在适当的时候调用我们注册的回调函数：handle_message当总线上出现消息时，以及 handle_keyboard当用户按下任意键时。

handle_message 函数没有什么新内容，只是当管道进入 PLAYING 状态时，它会调用该analyze_streams函数：

```c
/* Extract some metadata from the streams and print it on the screen */
static void analyze_streams (CustomData *data) {
  gint i;
  GstTagList *tags;
  gchar *str;
  guint rate;

  /* Read some properties */
  g_object_get (data->playbin, "n-video", &data->n_video, NULL);
  g_object_get (data->playbin, "n-audio", &data->n_audio, NULL);
  g_object_get (data->playbin, "n-text", &data->n_text, NULL);
```

正如注释中所述，此功能仅从媒体中收集信息并将其显示在屏幕上。视频、音频和字幕流的数量可直接通过 `video`、`audio` 和 `subtitle`n-video属性 获取。n-audion-text

```c
for (i = 0; i < data->n_video; i++) {
  tags = NULL;
  /* Retrieve the stream's video tags */
  g_signal_emit_by_name (data->playbin, "get-video-tags", i, &tags);
  if (tags) {
    g_print ("video stream %d:\n", i);
    gst_tag_list_get_string (tags, GST_TAG_VIDEO_CODEC, &str);
    g_print ("  codec: %s\n", str ? str : "unknown");
    g_free (str);
    gst_tag_list_free (tags);
  }
}
```

现在，对于每个数据流，我们需要检索其元数据。元数据以标签的形式存储在一个GstTagList结构体中，该结构体是一个由名称标识的数据片段列表。GstTagList可以使用 `get_database_name()` 函数检索与数据流关联的结构体，然后使用 `get_database_name ()` 等函数g_signal_emit_by_name()提取各个标签。gst_tag_list_get_*gst_tag_list_get_string()

    这种不太直观的标签列表检索方式称为动作信号。动作信号由应用程序发出，发送给特定元素，该元素随后执行操作并返回结果。它们的行为类似于动态函数调用，其中类的方法通过其名称（信号名称）而不是内存地址来标识。这些信号与常规信号一起列在文档中，并带有“Action”标签。playbin例如，请参阅[此处]。

playbin定义了 3 个用于检索元数据的操作信号： get-video-tags`@data`、` @video`get-audio-tags和`@text` get-text-tags。标签名称已标准化，完整列表可在GstTagList 文档中找到。在本例中，我们关注的是 GST_TAG_LANGUAGE_CODE流的 `@data` 及其类型GST_TAG_*_CODEC （音频、视频或文本）。

```c
g_object_get (data->playbin, "current-video", &data->current_video, NULL);
g_object_get (data->playbin, "current-audio", &data->current_audio, NULL);
g_object_get (data->playbin, "current-text", &data->current_text, NULL);
```

一旦我们提取了所有想要的元数据，我们就可以通过以下 3 个属性获取当前选定的流playbin： current-video，current-audio和current-text。

始终检查当前选定的流，切勿妄下断言，这一点非常重要。多种内部条件可能导致程序 playbin在不同的执行过程中表现不同。此外，流的排列顺序也可能因运行而异，因此检查元数据以识别特定流至关重要。

```c
/* Process keyboard input */
static gboolean handle_keyboard (GIOChannel *source, GIOCondition cond, CustomData *data) {
  gchar *str = NULL;

  if (g_io_channel_read_line (source, &str, NULL, NULL, NULL) == G_IO_STATUS_NORMAL) {
    int index = g_ascii_strtoull (str, NULL, 0);
    if (index < 0 || index >= data->n_audio) {
      g_printerr ("Index out of bounds\n");
    } else {
      /* If the input was a valid audio stream index, set the current audio stream */
      g_print ("Setting current audio stream to %d\n", index);
      g_object_set (data->playbin, "current-audio", index, NULL);
    }
  }
  g_free (str);
  return TRUE;
}
```

最后，我们允许用户切换正在运行的音频流。这个非常基本的功能只是从标准输入（键盘）读取一个字符串，将其解释为一个数字，并尝试设置 （之前我们只是读取的）current-audio属性。playbin

请注意，切换并非立即完成。部分先前解码的音频仍会在管道中传输，同时新的音频流才会激活并开始解码。延迟时间取决于容器中音频流的具体复用方式，以及 playbin其内部队列的长度（该长度取决于网络状况）。
